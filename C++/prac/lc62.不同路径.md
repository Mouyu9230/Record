##一个机器人位于一个 m x n 网格的左上角 （起始点在下图中标记为 “Start” ）。机器人每次只能向下或者向右移动一步。机器人试图达到网格的右下角（在下图中标记为 “Finish” ）。问总共有多少条不同的路径？

            vector<int> byd;

        run_it_back(m,n,0,0,all);

    int run_it_back(int wid,int len,int wpos,int lpos,int& all){
        if(wid-1==wpos&&len-1==lpos){
            all++;
            return 1;
        }
        if(wpos<wid){
            run_it_back(wid,len,wpos+1,lpos,all);
        }
        if(lpos<len){
            run_it_back(wid,len,wpos,lpos+1,all);
        }
        return 0;
    }
};

dp正确解
```
class Solution {
public:
    int uniquePaths(int m, int n) {
        vector<vector<int>> mapp;
        for(int i=0;i<m;i++){
            vector<int> byd;
            mapp.push_back(byd);
            for(int j=0;j<n;j++){
              mapp[i].push_back(0);
            }

        }
        for(int i=0;i<n;i++){
            mapp[0][i]=1;
        }
        for(int i=1;i<m;i++){
            mapp[i][0]=1;
        }
        for(int i=1;i<m;i++){
            for(int j=1;j<n;j++){
                mapp[i][j]=mapp[i-1][j]+mapp[i][j-1];
            }
        }

        return mapp[m-1][n-1];
        


    }
```
