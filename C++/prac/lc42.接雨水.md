暴力，对每个柱子前后方用max_element找最高的，两者中较短的那方减去柱子本身高度就是这根柱子上水量，超时
```
class Solution {
public:
    int trap(vector<int>& height) {
        int ans=0;
        int len=height.size();
        for(int i=1;i<len-1;i++){
            int maxminor=min(find_max(height,0,i-1),find_max(height,i+1,len-1));
            if(maxminor-height[i]>0){
                ans+=(maxminor-height[i]);
            }
        }
        return ans;
    }

    int find_max(vector<int>& height,int left,int right){
        auto byd=max_element(height.begin()+left,height.begin()+right+1);
        return *byd;
    }
};
```
在这个基础上优化的方法是采用维护两个最大值数组的方式来替代一次次的循环查找
```
class Solution {
public:
    int trap(vector<int>& height) {
        int ans=0;
        int len=height.size();

        vector<int> left_max;
        vector<int> right_max;

        int for_max=0;
        for(int i=0;i<len;i++){
            if(height[i]>for_max){
                for_max=height[i];
            }
            left_max.push_back(for_max);
        }

        for_max=0;

        for(int i=len-1;i>=0;i--){
            if(height[i]>for_max){
                for_max=height[i];
            }
            right_max.push_back(for_max);
        }
        reverse(right_max.begin(),right_max.end());



        for(int i=1;i<len-1;i++){
            int maxminor=min(left_max[i],right_max[i]);
            if(maxminor-height[i]>0){
                ans+=(maxminor-height[i]);
            }
        }
        return ans;
    }


};
```
