## 以数组 intervals 表示若干个区间的集合，其中单个区间为 intervals[i] = [starti, endi] 。请你合并所有重叠的区间，并返回 一个不重叠的区间数组，该数组需恰好覆盖输入中的所有区间 。
hot100  
我的解：映射到数组里再遍历，很直接的想法，但是因为输出刚需重叠而失败，解决方式是把数组翻了个倍来找到间隔的0来判断重叠与否  
```
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals){
        int max=0;
        int expanding=0;
        for(auto x:intervals){
            if(x[1]*2>max)max=x[1]*2;
        }

        vector<int> table(max+1,0);
        vector<vector<int>> ans;
        vector<int> for_push;

        for(auto x:intervals){
            for(int i=x[0]*2;i<=x[1]*2;i++){
                table[i]=1;
            }
        }

        for(int i=0;i<=max;i++){
            if(expanding==0&&table[i]==1){
                expanding=1;
                for_push.push_back(i/2);
            }

            if(expanding==1&&table[i]==0){
                expanding=0;
                for_push.push_back((i-1)/2);
                ans.push_back(for_push);
                for_push.clear();
            }

            if(expanding==1&&i==max&&table[i]==1){
                for_push.push_back(max/2);
                ans.push_back(for_push);
            }
        }

        return ans;
    }
};
```
挺铸币的，官方的方法更简单，排序完维护一个区间，新区间重叠了就更新，否则加入答案，转维护新区间，最后加入最后一个区间  
```
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals){
        sort(intervals.begin(),intervals.end());

        vector<vector<int>> ans;

        int left=intervals[0][0];
        int right=intervals[0][1];

        for(int i=1;i<intervals.size();i++){
            if(intervals[i][0]<=right){
                right=max(right,intervals[i][1]);
            }else{
                ans.push_back({left,right});
                left=intervals[i][0];
                right=intervals[i][1];
            }
        }

        ans.push_back({left,right});

        return ans;
    }
};
```
