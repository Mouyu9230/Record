### 给你一个任务数组 tasks ，其中 tasks[i] = [actuali, minimumi] ：actuali 是完成第 i 个任务 需要耗费 的实际能量。minimumi 是开始第 i 个任务前需要达到的最低能量。比方说，如果任务为 [10, 12] 且你当前的能量为 11 ，那么你不能开始这个任务。如果你当前的能量为 13 ，你可以完成这个任务，且完成它后剩余能量为 3 。你可以按照 任意顺序 完成任务。请你返回完成所有任务的 最少 初始能量。
5.12 daily  
简单的贪心，运用lambda来完成对依照vector中特定元素的sort  
```
class Solution {
public:
    int minimumEffort(vector<vector<int>>& tasks) {

        int size=tasks.size();

        vector<vector<int>> order;//每个v成员的两个int成员分别为actual与mini的差值和下标，依照前者sort,依照后者来更改原生task的顺序变为ordered_task
        vector<vector<int>> ordered_task;

        for(int i=0;i<size;i++){
            vector<int> byd;
            byd.push_back(tasks[i][1]-tasks[i][0]);//差
            byd.push_back(i);//下标
            order.push_back(byd);
        }
        sort(order.begin(), order.end(), [](const std::vector<int>& a, const std::vector<int>& b){//lambda部分，用到了vector字典序比较的特性，从而直接比vector本身就能完成对比第一个元素大小的任务
            return a>b; 
        });
        for(auto x:order){
            ordered_task.push_back(tasks[x[1]]);
        }

        int ans=0;
        int cur=0;
        


        for (auto x:ordered_task) {

            if(cur<x[1]) {
                int cha = x[1] - cur;
                ans += cha;
                cur = x[1];
            }

            cur -= x[0];
        }

        return ans;


    }
};
```
